ID ::= [a-zA-Z_][a-zA-Z0-9_]*
ENTERO ::= [0-9]+
FLOTANTE ::= [0-9]+\.[0-9]+
CARACTER ::= '[^']'
CADENA ::= "[^"]*"
TEXTO ::= .*

OP_SUMA    ::= "+"
OP_RESTA   ::= "-"
OP_MULT    ::= "*"
OP_DIV     ::= "/"
OP_DIV_ENT ::= "//"
OP_MOD     ::= "%"
OP_POT     ::= "^"
OP_INC     ::= "++"
OP_DEC     ::= "--"

OP_MENOR ::= "<"
OP_MAYOR ::= ">"
OP_MENIG ::= "<="
OP_MAYIG ::= ">="
OP_IGUAL ::= "=="
OP_DIFER ::= "!="

COMA ::= ","

BLOQUE_ABRE ::= "¿:"
BLOQUE_CIERRA ::= ":?"
INDICE_ABRE ::= "ʃ:"
INDICE_CIERRA ::= ":ʅ"
PAREN_ABRE ::= "є:"
PAREN_CIERRA ::= ":э"
ASIGNACION ::= "Ͱ"
FIN_SENT ::= "»"
COM_LINEA ::= "|"
COM_MULT_ABRE ::= "¡"
COM_MULT_CIERRA ::= "!"

OP_AND ::= "λ"
OP_OR  ::= "θ"
OP_NOT ::= "Σ"

<identificador> ::= ID
<tipo> ::= int | float | bool | char | str

<literal>   ::= <literalInt> | <literalFloat> | <literalBool> | <literalChar> | <literalStr>
<literalInt>    ::= ENTERO
<literalFloat>  ::= FLOTANTE
<literalBool>   ::= true | false
<literalChar>   ::= CARACTER
<literalStr>    ::= CADENA

<literalListInt>    ::= <literalInt> | <literalInt> COMA <literalListInt>
<literalListFloat>  ::= <literalFloat> | <literalFloat> COMA <literalListFloat>

<declaracion> ::= val <tipo> <identificador> FIN_SENT
                | val <tipo> <identificador> ASIGNACION <expresion> FIN_SENT

<declaracionList> ::= val int <identificador> FIN_SENT
                    | val float <identificador> FIN_SENT
                    | val int <identificador> ASIGNACION BLOQUE_ABRE <literalListInt> BLOQUE_CIERRA FIN_SENT
                    | val float <identificador> ASIGNACION BLOQUE_ABRE <literalListFloat> BLOQUE_CIERRA FIN_SENT

<exprArit> ::= <term> | <exprArit> OP_SUMA <term> | <exprArit> OP_RESTA <term>

<term> ::= <factor> | <term> OP_MULT <factor> | <term> OP_DIV <factor> | <term> OP_DIV_ENT <factor>

<factor> ::= <potencia> | <factor> OP_MOD <potencia>

<potencia> ::= <atomo> | <potencia> OP_POT <atomo>

<atomo> ::= <literal> | <identificador> | PAREN_ABRE <exprArit> PAREN_CIERRA
          | <negativo> | <unarioSuma> | <unarioResta> 
          | <llamadaFun> | <accesoList>

<accesoList> ::= <identificador> INDICE_ABRE <exprArit> INDICE_CIERRA

<negativo> ::= OP_RESTA <literalInt> | OP_RESTA <literalFloat>

<unarioSuma> ::= <identificador> OP_INC

<unarioResta> ::= <identificador> OP_DEC

<exprRel> ::= INDICE_ABRE <exprArit> INDICE_CIERRA <opRel> INDICE_ABRE <exprArit> INDICE_CIERRA
            | INDICE_ABRE <operandoEq> INDICE_CIERRA OP_IGUAL INDICE_ABRE <operandoEq> INDICE_CIERRA
            | INDICE_ABRE <operandoEq> INDICE_CIERRA OP_DIFER INDICE_ABRE <operandoEq> INDICE_CIERRA

<opRel> ::= OP_MENOR | OP_MAYOR | OP_MENIG | OP_MAYIG

<operandoEq> ::= <exprArit> | <literalBool> | <literalChar> | <identificador>

<exprLog> ::= PAREN_ABRE <exprLog> PAREN_CIERRA <opLog> PAREN_ABRE <exprLog> PAREN_CIERRA
            | OP_NOT PAREN_ABRE <exprLog> PAREN_CIERRA
            | <exprRel> | <literalBool> | <identificador>

<opLog> ::= OP_AND | OP_OR

<expresion> ::= <exprArit> | <exprRel> | <exprLog> 

<sentencia> ::= <declaracion> | <declaracionList> | <asignacion>
              | <expresion> FIN_SENT | <write> | <read>
              | <return> | <break>

<asignacion> ::= <identificador> ASIGNACION <expresion> FIN_SENT
               | <identificador> INDICE_ABRE <exprArit> INDICE_CIERRA ASIGNACION <expresion> FIN_SENT

<if> ::= if <condicion> BLOQUE_ABRE <bloque> BLOQUE_CIERRA <elifOpc> <elseOpc>

<elifOpc> ::= <elif> <elifOpc> |  ε
<elseOpc> ::= <else> | ε

<elif> ::= elif <condicion> BLOQUE_ABRE <bloque> BLOQUE_CIERRA
<else> ::= else BLOQUE_ABRE <bloque> BLOQUE_CIERRA

<condicion> ::= <exprLog> | <exprRel> | <literalBool> | <identificador>

<bloque> ::= <lineaBloque> | <lineaBloque> <bloque>

<lineaBloque> ::= <sentencia> | <estructuraControl> | <comentario>

<estructuraControl> ::= <if> | <while> | <for>

<while> ::= while <condicion> BLOQUE_ABRE <bloque> BLOQUE_CIERRA

<for> ::= for <condicionFor> BLOQUE_ABRE <bloque> BLOQUE_CIERRA

<condicionFor> ::= <declaracion> COMA <condicion> COMA <sentencia>

<return> ::= return <expresion> FIN_SENT

<break> ::= break FIN_SENT

<write> ::= write <paramWrite> FIN_SENT
<paramWrite> ::= <literalStr> | <literalInt> 
                | <literalBool> | <literalFloat> | <identificador>

<read> ::= read <identificador> FIN_SENT

<funcion> ::= fun <tipo> <identificador> <paramFun> BLOQUE_ABRE <bloque> BLOQUE_CIERRA

<params> ::= <tipo> <identificador> | <tipo> <identificador> COMA <params>

<paramFun> ::= PAREN_ABRE <params> PAREN_CIERRA | PAREN_ABRE PAREN_CIERRA

<llamadaFun> ::= <identificador> PAREN_ABRE <args> PAREN_CIERRA
               | <identificador> PAREN_ABRE PAREN_CIERRA

<args> ::= <expresion> | <expresion> COMA <args>

<principal> ::= void principal BLOQUE_ABRE <bloque> BLOQUE_CIERRA

<comentario> ::= COM_LINEA TEXTO | COM_MULT_ABRE TEXTO COM_MULT_CIERRA

<programa> ::= <globalesOpc> <funcionesOpc> <principal>

<globalesOpc> ::= <declaracion> <globalesOpc> 
                | <declaracionList> <globalesOpc>
                | <comentario> <globalesOpc>
                | ε

<funcionesOpc> ::= <funcion> <funcionesOpc> | ε
