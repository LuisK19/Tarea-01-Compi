<identificador> ::= ID
<tipo> ::= int | float | bool | char | str

<literal>   ::= <literalInt> | <literalFloat> | <literalBool> | <literalChar> | <literalStr>
<literalInt>    ::= ENTERO
<literalFloat>  ::= FLOTANTE
<literalBool>   ::= true | false
<literalChar>   ::= CARACTER
<literalStr>    ::= CADENA

<literalListInt>    ::= <literalInt> | <literalInt> , <literalListInt>
<literalListFloat>  ::= <literalFloat> | <literalFloat> , <literalListFloat>

<declaracion>   ::= val <tipo> <identificador> »
                | val <tipo> <identificador> Ͱ <expresion> »

<declaracionList>   ::= val int <identificador> »
                    | val float <identificador> »
                    | val int <identificador> Ͱ ¿: <literalListInt> :? »
                    | val float <identificador> Ͱ ¿: <literalListFloat> :? »

<exprArit>  ::= <term> | <exprArit> + <term> | <exprArit> - <term>

<term>  ::= <factor> | <term> * <factor> | <term> / <factor> | <term> // <factor>

<factor>    ::= <potencia> | <factor> % <potencia>

<potencia>  ::= <atomo> | <potencia> ^ <atomo>

<atomo>   ::= <literal> | <identificador> | є: <exprArit> :э
          | <negativo> | <unarioSuma> | <unarioResta> 
          | <llamadaFun> | <accesoList>

<accesoList> ::= <identificador> ʃ: <exprArit> :ʅ

<negativo> ::= - <literalInt> | - <literalFloat> 

<unarioSuma> ::= <identificador> ++ 

<unarioResta> ::= <identificador> --

<exprRel> ::= ʃ: <exprArit> :ʅ <opRel> ʃ: <exprArit> :ʅ  
            | ʃ: <operandoEq> :ʅ == ʃ: <operandoEq> :ʅ 
            | ʃ: <operandoEq> :ʅ != ʃ: <operandoEq> :ʅ

<opRel> ::= < | > | <= | >= 

<operandoEq> ::= <exprArit> | <literalBool> | <literalChar> | <identificador>

<exprLog> ::= є: <exprLog> :э <opLog> є: <exprLog> :э
            | Σ є: <exprLog> :э
            | <exprRel> | <literalBool> | <identificador>

<opLog> ::= λ | θ 

<expresion> ::= <exprArit> | <exprRel> | <exprLog> 

<sentencia> ::= <declaracion> | <declaracionList> | <asignacion> 
              | <expresion> » | <write> | <read>
              | <return> | <break>
